===============================================================================
 THARAK / SYSADMIN WINDOWS INCIDENT RESPONSE COLLECTOR
===============================================================================

 PURPOSE
 -------
 Read-only forensic collection and correlation for a suspected Windows
 compromise involving:

   - AD credentials: tharak / sysadmin
   - Lateral movement
   - Privileged execution
   - Persistence
   - Scheduled tasks
   - Services
   - Registry modification

   - D: volume activity

 DEFAULT INCIDENT WINDOW
 -----------------------
 04-Oct-2026 03:00 IST -> 04-Oct-2026 11:00 IST

 MODES
 -----
 Local case:

   .\IR.ps1

 Export case to share:

   .\IR.ps1 -Export "\\server\share\IR"

 Optional:

   .\IR.ps1 -Export "\\server\share\IR" -HashFiles

 IMPORTANT
 ---------
 This script is READ ONLY.

 It does NOT:
   - kill processes
   - disable accounts
   - delete persistence
   - clear event logs
   - change registry
   - enable/disable BitLocker
   - modify firewall
   - alter security policy

 LIMITATIONS
 -----------
 Exact process command lines require command-line auditing / 4688 command
 line collection to have been enabled.

 Registry modification events require appropriate object auditing.

 Historical network connections cannot be reconstructed from live
 Get-NetTCPConnection output if they were not logged elsewhere.

 For strongest enterprise-wide attribution, collect the same time window
 from Domain Controllers as well.

===============================================================================

function Write-IRStep {
    param(
        [string]$Message
    )

    $line = "[{0}] [INFO] {1}" -f `
        (Get-Date -Format "yyyy-MM-dd HH:mm:ss"),
        $Message

    Write-Host $line

    try {
        Add-Content `
            -LiteralPath $LogFile `
            -Value $line `
            -Encoding UTF8
    }
    catch {}
}

function Write-IRError {
    param(
        [string]$Collector,
        [string]$Message
    )

    $obj = [PSCustomObject]@{
        Time      = Get-Date
        Computer  = $HostName
        Collector = $Collector
        Error     = $Message
    }

    [void]$Errors.Add($obj)

    try {
        Add-Content `
            -LiteralPath $LogFile `
            -Value ("[{0}] [WARN] {1} :: {2}" -f `
                (Get-Date -Format "yyyy-MM-dd HH:mm:ss"),
                $Collector,
                $Message) `
            -Encoding UTF8
    }
    catch {}
}

function Add-Finding {
    param(
        [ValidateSet("CRITICAL","HIGH","MEDIUM","LOW")]
        [string]$Severity,

        [string]$Category,

        [string]$Indicator,

        [string]$Time,

        [string]$Account,

        [string]$Source,

        [string]$Target,

        [string]$Process,

        [string]$CommandLine,

        [string]$Evidence
    )

    $obj = [PSCustomObject]@{
        Severity    = $Severity
        Time        = $Time
        Computer    = $HostName
        Category    = $Category
        Indicator   = $Indicator
        Account     = $Account
        Source      = $Source
        Target      = $Target
        Process     = $Process
        CommandLine = $CommandLine
        Evidence    = $Evidence
    }

    [void]$Findings.Add($obj)

    # Console only shows findings, not every event.
    $color = switch ($Severity) {
        "CRITICAL" { "Magenta" }
        "HIGH"     { "Red" }
        "MEDIUM"   { "Yellow" }
        default    { "Cyan" }
    }

    Write-Host `
        ("[{0}] [{1}] [{2}] {3}" -f `
            (Get-Date -Format "HH:mm:ss"),
            $Severity,
            $Category,
            $Indicator) `
        -ForegroundColor $color

    if ($Account) {
        Write-Host "       Account : $Account" -ForegroundColor DarkGray
    }

    if ($Source) {
        Write-Host "       Source  : $Source" -ForegroundColor DarkGray
    }

    if ($Target) {
        Write-Host "       Target  : $Target" -ForegroundColor DarkGray
    }

    if ($Process) {
        Write-Host "       Process : $Process" -ForegroundColor DarkGray
    }

    if ($CommandLine) {
        Write-Host "       Command : $CommandLine" -ForegroundColor DarkGray
    }

    if ($Evidence) {
        Write-Host "       Evidence: $Evidence" -ForegroundColor DarkGray
    }
}

function Save-Csv {
    param(
        [object]$Data,
        [string]$Name
    )

    try {

        $path = Join-Path $CaseRoot $Name

        $items = @($Data)

        if ($items.Count -eq 0) {

            [PSCustomObject]@{
                Status = "No records collected"
            } |
            Export-Csv `
                -LiteralPath $path `
                -NoTypeInformation `
                -Encoding UTF8

            return
        }

        $items |
            Export-Csv `
                -LiteralPath $path `
                -NoTypeInformation `
                -Encoding UTF8
    }
    catch {

        Write-IRError `
            "CSV:$Name" `
            $_.Exception.Message
    }
}

function Save-Text {
    param(
        [string]$Text,
        [string]$Name
    )

    try {

        Set-Content `
            -LiteralPath (Join-Path $CaseRoot $Name) `
            -Value $Text `
            -Encoding UTF8
    }
    catch {

        Write-IRError `
            "TEXT:$Name" `
            $_.Exception.Message
    }
}

# ============================================================================
# EVENT XML PARSER
# ============================================================================

function Get-EventDataHashtable {
    param(
        [System.Diagnostics.Eventing.Reader.EventRecord]$Event
    )

    $result = @{}

    try {

        [xml]$xml = $Event.ToXml()

        $nodes = @(
            $xml.Event.EventData.Data
        )

        foreach ($node in $nodes) {

            if ($null -ne $node.Name) {

                $name = [string]$node.Name
                $value = [string]$node.'#text'

                $result[$name] = $value
            }
        }
    }
    catch {}

    return $result
}

function Get-EventValue {
    param(
        [hashtable]$Data,
        [string[]]$Names
    )

    foreach ($name in $Names) {

        if ($Data.ContainsKey($name)) {

            if (-not [string]::IsNullOrWhiteSpace(
                [string]$Data[$name]
            )) {
                return [string]$Data[$name]
            }
        }
    }

    return ""
}

# ============================================================================
# PIVOT TESTING
# ============================================================================

function Test-Pivot {
    param(
        [string]$Text
    )

    if ([string]::IsNullOrWhiteSpace($Text)) {
        return @()
    }

    $matched = New-Object System.Collections.ArrayList

    foreach ($user in $PivotUsers) {

        if ([string]::IsNullOrWhiteSpace($user)) {
            continue
        }

        try {

            if ($Text -match "(?i)\b$([regex]::Escape($user))\b") {
                [void]$matched.Add($user)
            }
        }
        catch {}
    }

    return @($matched)
}

function Is-SuspiciousCommand {
    param(
        [string]$Command
    )

    if ([string]::IsNullOrWhiteSpace($Command)) {
        return $false
    }

    return (
        $Command -match "(?i)\bmanage-bde(?:\.exe)?\b" -or
        $Command -match "(?i)\b(powershell|pwsh)\b" -or
        $Command -match "(?i)\b(mshta|wscript|cscript|rundll32|regsvr32|certutil|bitsadmin)\b" -or
        $Command -match "(?i)(EncodedCommand|FromBase64String|DownloadString|Invoke-WebRequest|Invoke-RestMethod)" -or
        $Command -match "(?i)\b(schtasks|sc\.exe|wmic|winrs|psexec)\b" -or
        $Command -match "(?i)\b(net\s+use|net\s+user|net\s+localgroup)\b"
    )
}

function Is-BitLockerCommand {
    param(
        [string]$Text
    )

    if ([string]::IsNullOrWhiteSpace($Text)) {
        return $false
    }

    return (
        $Text -match "(?i)\bmanage-bde(?:\.exe)?\b" -or
        $Text -match "(?i)\b(BitLocker|BitLockerVolume)\b" -or
        $Text -match "(?i)\bEnable-BitLocker\b" -or
        $Text -match "(?i)\bAdd-BitLockerKeyProtector\b" -or
        $Text -match "(?i)\bResume-BitLocker\b" -or
        $Text -match "(?i)\bBackup-BitLockerKeyProtector\b"
    )
}

catch {
    Write-IRError "LocalUsers" $_.Exception.Message

try {

    $adminMembers = @()

    $group = [ADSI]"WinNT://$HostName/Administrators,group"

    foreach ($member in @($group.psbase.Invoke("Members"))) {

        $name = ""
        $domain = ""

        try {
            $name = $member.GetType().InvokeMember(
                "Name",
                "GetProperty",
                $null,
                $member,
                $null
            )
        }
        catch {}

        try {
            $domain = $member.GetType().InvokeMember(
                "Domain",
                "GetProperty",
                $null,
                $member,
                $null
            )
        }
        catch {}

        $adminMembers += [PSCustomObject]@{
            Domain = $domain
            Name   = $name
        }

        if ($name -ieq $SuspectUser -or
            $name -ieq "sysadmin") {

            Add-Finding `
                -Severity "HIGH" `
                -Category "PRIVILEGE" `
                -Indicator "Investigation account is local administrator" `
                -Time (Get-Date).ToString("o") `
                -Account "$domain\$name" `
                -Source "" `
                -Target $HostName `
                -Process "" `
                -CommandLine "" `
                -Evidence "Local Administrators group"
        }
    }

    Save-Csv `
        $adminMembers `
        "LOCAL_ADMINISTRATORS.csv"
}
catch {
    Write-IRError "LocalAdministrators" $_.Exception.Message
}

Write-IRStep "03/20 Collecting Active Directory user inventory"

try {

    Import-Module ActiveDirectory -ErrorAction Stop

    $adUsers = @(
        Get-ADUser `
            -Filter * `
            -Properties `
                Enabled,
                LastLogonDate,
                PasswordLastSet,
                PasswordNeverExpires,
                AdminCount,
                LockedOut,
                DistinguishedName,
                MemberOf,
                WhenCreated,
                WhenChanged |
        Select-Object `
            SamAccountName,
            UserPrincipalName,
            Enabled,
            LockedOut,
            AdminCount,
            LastLogonDate,
            PasswordLastSet,
            PasswordNeverExpires,
            WhenCreated,
            WhenChanged,
            DistinguishedName,
            @{N="MemberOf";E={
                if ($_.MemberOf) {
                    ($_.MemberOf -join " | ")
                }
                else {
                    ""
                }
            }}
    )

    Save-Csv `
        $adUsers `
        "AD_USERS.csv"

    foreach ($adUser in $adUsers) {

        if ($adUser.SamAccountName -ieq $SuspectUser -or
            $adUser.SamAccountName -ieq "sysadmin") {

            Add-Finding `
                -Severity "HIGH" `
                -Category "AD_IDENTITY" `
                -Indicator "Incident pivot account present in AD" `
                -Time $adUser.WhenChanged.ToString("o") `
                -Account $adUser.SamAccountName `
                -Source "" `
                -Target $HostName `
                -Process "" `
                -CommandLine "" `
                -Evidence "Enabled=$($adUser.Enabled); AdminCount=$($adUser.AdminCount); LastLogon=$($adUser.LastLogonDate)"
        }
    }
}
catch {
    Write-IRError `
        "ActiveDirectoryInventory" `
        "AD module unavailable or query failed: $($_.Exception.Message)"
}
